import { test, expect, type Page } from '@playwright/test'
import { bootApp, seedCode } from './_appBoot'

// #1731 exploratory probe — NOT a gate. Observes the synth-lane envelope in the
// real app: does it render while stopped, how long, what it looks like.

const attr = (page: Page, name: string) =>
  page.evaluate((n) => document.querySelector(`[${n}]`)?.getAttribute(n) ?? null, name)

test('observe: a stopped synth lane gets an envelope', async ({ page }) => {
  test.setTimeout(120_000)
  const logs: string[] = []
  page.on('console', (m) => {
    if (m.type() === 'error' || m.type() === 'warning') logs.push(`${m.type()}: ${m.text()}`)
  })
  await bootApp(page, { drawer: { tabId: 'musical-timeline', height: 360 } })
  const t0 = Date.now()
  await seedCode(page, process.env.PROBE_CODE ?? 'setcps(0.5)\n$: note("c3*4").s("sawtooth").gain("<0.15 1>")\n$: s("bd*4")')
  const seen: string[] = []
  let fresh = ''
  for (let i = 0; i < 120; i++) {
    const e = (await attr(page, 'data-full-song-envelopes')) ?? ''
    const r = (await attr(page, 'data-full-song-envelope-rendering')) ?? ''
    const line = `${Date.now() - t0}ms envelopes="${e}" rendering="${r}"`
    if (seen[seen.length - 1]?.split('ms ')[1] !== line.split('ms ')[1]) seen.push(line)
    if (r !== '' && !e.includes(':fresh') && !(globalThis as { __shot?: boolean }).__shot) {
      ;(globalThis as { __shot?: boolean }).__shot = true
      await page.locator('[data-full-song-canvas]').screenshot({ path: `test-results/_1731-canvas-${process.env.PROBE_TAG ?? 'a'}-before.png` })
    }
    if (e.includes(':fresh')) {
      fresh = e
      break
    }
    await page.waitForTimeout(50)
  }
  console.log(seen.join('\n'))
  const lanes = await page.evaluate(() =>
    Array.from(document.querySelectorAll('[data-full-song-lane]')).map((l) => l.getAttribute('data-full-song-lane')),
  )
  console.log('lanes', lanes, 'period', await attr(page, 'data-full-song-period'))
  await page.locator('[data-full-song-canvas]').screenshot({ path: `test-results/_1731-canvas-${process.env.PROBE_TAG ?? 'a'}.png` })
  await page.locator('[data-bottom-panel="root"]').screenshot({ path: `test-results/_1731-stopped-${process.env.PROBE_TAG ?? 'a'}.png` })
  console.log(logs.slice(0, 20).join('\n'))
  expect(fresh).not.toBe('')
})
