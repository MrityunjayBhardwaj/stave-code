import { test } from '@playwright/test'
import { bootApp, seedCode } from './_appBoot'
// #1731 probe (not a gate): how many 1 s intervals are alive on a stopped
// document whose sample lane never plays.
test('observe: decode re-check timers on a stopped sample lane', async ({ page }) => {
  await page.addInitScript(() => {
    const w = window as unknown as { __live1s: Set<number> }
    w.__live1s = new Set()
    const si = window.setInterval.bind(window)
    const ci = window.clearInterval.bind(window)
    ;(window as unknown as { setInterval: typeof setInterval }).setInterval = ((fn: TimerHandler, ms?: number, ...a: unknown[]) => {
      const id = si(fn, ms, ...a) as unknown as number
      if (ms === 1000) w.__live1s.add(id)
      return id
    }) as typeof setInterval
    ;(window as unknown as { clearInterval: typeof clearInterval }).clearInterval = ((id?: number) => {
      w.__live1s.delete(id as number)
      ci(id)
    }) as typeof clearInterval
  })
  await bootApp(page, { drawer: { tabId: 'musical-timeline', height: 360 } })
  await seedCode(page, 'setcps(0.5)\n$: s("bd*4")')
  await page.waitForFunction(() => document.querySelectorAll('[data-full-song-lane]').length === 1, null, { timeout: 20_000 })
  const n = await page.evaluate(async () => {
    const w = window as unknown as { __live1s: Set<number> }
    const at: number[] = []
    for (let i = 0; i < 4; i++) { await new Promise((r) => setTimeout(r, 1500)); at.push(w.__live1s.size) }
    return at
  })
  console.log(`[timer ${process.env.PROBE_TAG}] live 1 s intervals over 6 s: ${n.join(',')}`)
  const peaks = () => page.evaluate(() => document.querySelector('[data-full-song-lane]') != null)
  await peaks()
  await page.locator('.monaco-editor').first().click()
  await page.keyboard.press(process.platform === 'darwin' ? 'Meta+Enter' : 'Control+Enter')
  const playing: number[] = []
  for (let i = 0; i < 12; i++) {
    await page.waitForTimeout(250)
    playing.push(await page.evaluate(() => (window as unknown as { __live1s: Set<number> }).__live1s.size))
  }
  console.log(`[timer ${process.env.PROBE_TAG}] while playing, every 250 ms: ${playing.join(',')}`)
})
